Quick copy-paste

Use these in your code or terminal.

curl "https://api.sitetrace.it.com/api/shot?url=https://example.com" \
  -H "Authorization: Bearer YOUR_KEY" \
  --output shot.png
const r = await fetch('https://api.sitetrace.it.com/api/shot?url=https://example.com', {
  headers: { 'Authorization': 'Bearer YOUR_KEY' }
});
const blob = await r.blob();
import requests
r = requests.get(
  'https://api.sitetrace.it.com/api/shot',
  params={'url': 'https://example.com'},
  headers={'Authorization': 'Bearer YOUR_KEY'},
)
with open('shot.png', 'wb') as f:
    f.write(r.content)
